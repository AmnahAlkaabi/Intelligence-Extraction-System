# Aggregates the grouped files under requirements/ so local dev can still
# run a single `pip install -r requirements.txt`. The Docker build installs
# each group in its own layer instead (see Dockerfile) so a mid-install
# failure only costs a retry of that one group, not the whole set.
#
# torch/torchvision are NOT included here -- install the CPU-only build
# first (avoids pulling several GB of unused NVIDIA CUDA packages):
#   pip install torch==2.5.1 torchvision==0.20.1 --index-url https://download.pytorch.org/whl/cpu
-r requirements/core.txt
-r requirements/llm.txt
-r requirements/embeddings.txt
-r requirements/graph.txt
-r requirements/pdf.txt
-r requirements/ocr.txt
-r requirements/office.txt
-r requirements/web.txt
-r requirements/data.txt
-r requirements/utils.txt
-r requirements/test.txt
